- Based on: [KamaCoder **108. 多余的边**](KamaCoder%20108%20%E5%A4%9A%E4%BD%99%E7%9A%84%E8%BE%B9%203332cf069c0880219193dd27a0f69df7.md)
- 2 data to store:
    - *indegree* for each node
    - all *edges*
- 3 cases for redundent edge:
    - A node has 2 indgree, then one of its edges should be reomved
        - Both 2 edges are removable(form a tree), latter one first.
        - Only one edge is removable, after removing the irremovable edge, **rest will form a circle**(can detect by union-find set).
    - No nodes has 2 indgree, indicate there **exist a circle** in graph
- 2 functions:
    - `isTreeAfterRemoveEdge(e)`
        - Add all edges except detected one to union-find set， if all edges added successfully, return true, else false/
    - `decircleEdge(edges)`
        - Add all edges to union-find set, if an edge failed, print it out.

# Solution

```java
import java.util.*;

class UnionFind {
    int[] parent;

    public UnionFind(int n) {
        parent = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            parent[i] = i;
        }
    }

    public int find(int x) {
        if (parent[x] == x) return x;
        return parent[x] = find(parent[x]);
    }

    public boolean isConnected(int u, int v) {
        return find(u) == find(v);
    }

    public boolean union(int u, int v) {
        u = find(u);
        v = find(v);

        if (u == v) return false;
        parent[v] = u;
        return true;
    }
}

public class Main {
    static int[][] edges;
    static int n;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        n = in.nextInt();
        edges = new int[n][2];
        int[] inDegree = new int[n + 1];

        // store edges, count inDegree
        for (int i = 0; i < n; i++) {
            edges[i][0] = in.nextInt();
            edges[i][1] = in.nextInt();
            inDegree[edges[i][1]]++;
        }

        int[] res = new int[2];
        List<Integer> duoEdges = new ArrayList<>(); // edge index

        // find: edges ->node(2-in-degree), later edge first check
        for (int i = n - 1; i >= 0; i--) {
            if (inDegree[edges[i][1]] == 2) duoEdges.add(i); 
        }

        if (duoEdges.size() == 2) {
            int e1 = duoEdges.get(0);
            int e2 = duoEdges.get(1);

            // removablility
            if (isTreeAfterRemove(duoEdges.get(0))) { 
                System.out.print(edges[e1][0] + " " + edges[e1][1]);
            } else {
                System.out.print(edges[e2][0] + " " + edges[e2][1]);
            }

            return;
        }

        decircleEdge();
    }

    static boolean isTreeAfterRemove(int e) {
        UnionFind uf = new UnionFind(n);

        for (int i = 0; i < n; i++) {
            if (i == e) continue;
            if (!uf.union(edges[i][0], edges[i][1])) return false;
        }

        return true;
    }

    static void decircleEdge() {
        UnionFind uf = new UnionFind(n);

        for (int i = 0; i < n; i++) {
            int n1 = edges[i][0];
            int n2 = edges[i][1];

            if (!uf.union(n1, n2)) {
                System.out.print(n1 + " " + n2);
            }
        }
    }
}
```
