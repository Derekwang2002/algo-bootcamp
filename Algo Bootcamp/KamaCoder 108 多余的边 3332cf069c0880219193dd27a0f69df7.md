# Thought

- If two nodes has same parent nodes, then the edge between them is a redundent edge(form a circle)

# Solution

```java
import java.util.*;

class UnionFind {
    int[] parent;

    public UnionFind(int n) {
        parent = new int[n + 1];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }
    }

    public int find(int x) {
        if (parent[x] == x) return x;
        return parent[x] = find(parent[x]);
    }

    public boolean isConnected(int u, int v) {
        return find(u) == find(v);
    }

    public void union(int u, int v) {
        u = find(u);
        v = find(v);

        if (u == v) return;
        parent[v] = u;
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        UnionFind uf = new UnionFind(n);

        int[] edge = new int[2];

        while (in.hasNext()) {
            int u = in.nextInt();
            int v = in.nextInt();

            if (uf.isConnected(u, v)) {
                edge[0] = u;
                edge[1] = v;
            }

            uf.union(u, v);
        }

        System.out.print(edge[0] + " " + edge[1]);
    }
}
```
