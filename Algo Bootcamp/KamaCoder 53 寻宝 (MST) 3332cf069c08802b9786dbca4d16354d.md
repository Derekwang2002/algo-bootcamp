# Thought

## Prim

- **Prim MST algo**: [Prim: Greedy, add a nearest node(not in MST) into MST at each step, then renew rest nodes’ dist to MST](https://app.notion.com/p/Prim-Greedy-add-a-nearest-node-not-in-MST-into-MST-at-each-step-then-renew-rest-nodes-dist-to-M-3332cf069c088003b6e4cdcf6fae3344?pvs=21)
- 3 steps:
    - find nearest node to MST
    - add that node into MST
    - relaxation
- Details: when relaxing, don’t relax on the nodes already in MST!
- Code optimization:
    - Use boolean[] to record visited node(in MST)
    - Use List<int[]>[] to record graph info (`graph[i] = [to, weight]`)

## Kruskal

- **Kruskal MST algo:** [Kruskal:  Greedy, sort edge by weight, adding from minimum edge, if the new edge forms a circle, skip it](https://app.notion.com/p/Kruskal-Greedy-sort-edge-by-weight-adding-from-minimum-edge-if-the-new-edge-forms-a-circle-ski-3332cf069c0880f0a0c5d5b70d8f9e41?pvs=21)
    - Store edges: class
- Detect circle: union-find set

## Prim

```java
import java.util.*;

public class Main {
    static int[] minDist;
    static Map<Set<Integer>, Integer> edges;
    static List<Integer>[] vertices;
    static Set<Integer> mst;

    public static void main(String[] args) { 
        Scanner in = new Scanner(System.in);

        int v = in.nextInt();
        int e = in.nextInt();
        edges = new HashMap<>();
        vertices = new List[v + 1];

        for (int i = 1; i <= v; i++) {
            vertices[i] = new ArrayList<>();
        }

        // min distance to mst
        minDist = new int[v + 1];
        Arrays.fill(minDist, 10001);
        minDist[0] = 0;
        minDist[1] = 0;

        while (in.hasNext()) {
            int v1 = in.nextInt();
            int v2 = in.nextInt();
            int val = in.nextInt();

            vertices[v1].add(v2);
            vertices[v2].add(v1);
            edges.put(Set.of(v1, v2), val);
        }

        // mst set
        mst = new HashSet<>();        
        mst.add(1);
        relax(1);

        while (mst.size() != v) {
            int gap = 10001;
            int next = 0;

            for (int i = 2; i <= v; i++) {
                if (!mst.contains(i) && minDist[i] < gap) {
                    gap = minDist[i];
                    next = i;
                }
            }

            mst.add(next);
            relax(next);
        }

        int sum = 0;
        for (int i : minDist) {
            sum += i;
        }

        System.out.print(sum);
    }

    static void relax(int v) {
        for (int i : vertices[v]) {
            if (mst.contains(i)) continue;
            minDist[i] = Math.min(minDist[i], edges.get(Set.of(v, i)));
        }
    }
}
```

### Optimized prim

```java
import java.util.*;

public class Main {
    static List<int[]>[] graph; // [to, weight]

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int v = in.nextInt();
        int e = in.nextInt();

        graph = new List[v + 1];
        for (int i = 1; i <= v; i++) {
            graph[i] = new ArrayList<>();
        }

        // 建图
        for (int i = 0; i < e; i++) {
            int u = in.nextInt();
            int w = in.nextInt();
            int cost = in.nextInt();

            graph[u].add(new int[]{w, cost});
            graph[w].add(new int[]{u, cost});
        }

        // Prim
        boolean[] visited = new boolean[v + 1];
        int[] minDist = new int[v + 1];
        Arrays.fill(minDist, Integer.MAX_VALUE);

        minDist[1] = 0;
        int res = 0;

        for (int i = 1; i <= v; i++) {
            int u = -1;

            // 找当前最小的点
            for (int j = 1; j <= v; j++) {
                if (!visited[j] && (u == -1 || minDist[j] < minDist[u])) {
                    u = j;
                }
            }

            // 不连通
            if (minDist[u] == Integer.MAX_VALUE) {
                System.out.println(-1);
                return;
            }

            visited[u] = true;
            res += minDist[u];

            // 更新邻居
            for (int[] edge : graph[u]) {
                int to = edge[0];
                int weight = edge[1];

                if (!visited[to] && weight < minDist[to]) {
                    minDist[to] = weight;
                }
            }
        }

        System.out.println(res);
    }
}
```

## Kruskal

```java
import java.util.*;

class UnionFind {
    int[] parent;

    public UnionFind(int n) {
        parent = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            parent[i] = i;
        }
    }

    public int find(int v) {
        if (parent[v] == v) return v;
        return parent[v] = find(parent[v]);
    }

    public boolean isConnected(int u, int v) {
        return find(u) == find(v);
    }

    public void union(int u, int v) {
        u = find(u);
        v = find(v);

        if (u == v) return;
        parent[v] = u;
    }
}

class Edge {
    int v1, v2, w;

    public Edge(int v1, int v2, int weight) {
        this.v1 = v1;
        this.v2 = v2;
        w = weight;
    }
}

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int v = in.nextInt();
        int e = in.nextInt();

        Edge[] edges = new Edge[e];
        UnionFind uf = new UnionFind(v);

        for (int i = 0; i < e; i++) {
            int v1 = in.nextInt();
            int v2 = in.nextInt();
            int w = in.nextInt();

            edges[i] = new Edge(v1, v2, w);
        }

        Arrays.sort(edges, Comparator.comparingInt(edge -> edge.w));

        int res = 0;
        for (Edge edge : edges) {
            int v1 = edge.v1;
            int v2 = edge.v2;

            if (uf.isConnected(v1, v2)) continue;
            uf.union(v1, v2);
            res += edge.w;
        }

        System.out.print(res);
    }
}
```
