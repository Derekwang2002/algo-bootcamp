现有一个 N × M 的矩阵，每个单元格包含一个数值，这个数值代表该位置的相对高度。矩阵的左边界和上边界被认为是第一组边界，而矩阵的右边界和下边界被视为第二组边界。

矩阵模拟了一个地形，当雨水落在上面时，水会根据地形的倾斜向低处流动，但只能从较高或等高的地点流向较低或等高并且相邻（上下左右方向）的地点。我们的目标是确定那些单元格，从这些单元格出发的水可以达到第一组边界和第二组边界。

- input: 第一行包含两个整数 N 和 M，分别表示矩阵的行数和列数。后续 N 行，每行包含 M 个整数，表示矩阵中的每个单元格的高度。
    
    ```
    5 5
    1 3 1 2 4
    1 2 1 3 2
    2 4 7 2 1
    4 5 6 1 1
    1 4 1 2 1
    ```
    
- output: 输出共有多行，每行输出两个整数，用一个空格隔开，表示可达第一组边界和第二组边界的单元格的坐标，输出顺序任意。
    
    ```
    0 4
    1 3
    2 2
    3 0
    3 1
    3 2
    4 0
    4 1
    ```
    
</aside>

# Thought

- **Solution 1:** traverse map once and do dfs for each node, if it can reach both borders, print it out
    - trick: use int marks to determine if a node can reach two borders (low readability)
    - slower approach → $*O((VE)^2)*$
- **Solution 2(reverse):** imagine two traverse spread from each border, if a node is reachable in both traverse, print it out
    - more efficient approach → $*O(VE)*$

# Solution

```java
import java.util.*;

public class Main {
    static int[][] map;
    static int[][] visited;
    static int[][] directions = {{-1, 0}, {0, -1}, {1, 0}, {0, 1}};

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int rows = in.nextInt();
        int cols = in.nextInt();
        map = new int[rows][cols];
        visited = new int[rows][cols];

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) map[i][j] = in.nextInt();
        }

        int vID = 0;
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                vID++;
                if (dfs(i, j, vID) == 3) System.out.println(i + " " + j);
            }
        }
    }

    private static int dfs(int r, int c, int vID) {
        int res = 0;
        visited[r][c] = vID;

        for (int[] d : directions) {
            int newR = r + d[0];
            int newC = c + d[1];

            if (newR < 0 || newC < 0) res += 1;
            else if (newR > map.length - 1 || newC > map[0].length - 1) res += -3;
            else if (map[newR][newC] <= map[r][c] && visited[newR][newC] != vID) {
                int child = dfs(newR, newC, vID);
                if (child == 3) return 3;
                res += child;
            }

            if (res == -1 || res == -2 || res == -5) return 3;
        }

        if (res > 0) return 1;
        if (res < 0) return -3;

        return res;
    }
}
```

### Carl’s (reverse check)

```java
import java.util.*;

public class Main {
    static int n, m;
    static int[][] grid;
    static int[][] dir = {{-1, 0}, {0, -1}, {1, 0}, {0, 1}};

    static void dfs(boolean[][] visited, int x, int y) {
        if (visited[x][y]) return;
        visited[x][y] = true;

        for (int i = 0; i < 4; i++) {
            int nx = x + dir[i][0];
            int ny = y + dir[i][1];
            if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
            if (grid[x][y] > grid[nx][ny]) continue; // traverse low → high
            dfs(visited, nx, ny);
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        grid = new int[n][m];

        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                grid[i][j] = in.nextInt();

        boolean[][] firstBorder = new boolean[n][m];
        boolean[][] secondBorder = new boolean[n][m];

        // DFS from left column (firstBorder) and right column (secondBorder)
        for (int i = 0; i < n; i++) {
            dfs(firstBorder, i, 0);
            dfs(secondBorder, i, m - 1);
        }

        // DFS from top row (firstBorder) and bottom row (secondBorder)
        for (int j = 0; j < m; j++) {
            dfs(firstBorder, 0, j);
            dfs(secondBorder, n - 1, j);
        }

        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                if (firstBorder[i][j] && secondBorder[i][j])
                    System.out.println(i + " " + j);
    }
}
```

```java
import java.util.*;

public class Main {
    static int n, m;
    static int[][] grid;
    static int[][] dir = {{-1, 0}, {0, -1}, {1, 0}, {0, 1}};

    static void bfs(boolean[][] visited, Queue<int[]> queue) {
        while (!queue.isEmpty()) {
            int[] cur = queue.poll();
            for (int[] d : dir) {
                int nx = cur[0] + d[0];
                int ny = cur[1] + d[1];
                if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
                if (visited[nx][ny]) continue;
                if (grid[nx][ny] < grid[cur[0]][cur[1]]) continue; // only go uphill
                visited[nx][ny] = true;
                queue.offer(new int[]{nx, ny});
            }
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        grid = new int[n][m];

        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                grid[i][j] = in.nextInt();

        boolean[][] firstBorder = new boolean[n][m];
        boolean[][] secondBorder = new boolean[n][m];
        Queue<int[]> queue1 = new LinkedList<>();
        Queue<int[]> queue2 = new LinkedList<>();

        // Seed left column + top row into queue1
        for (int i = 0; i < n; i++) { firstBorder[i][0] = true; queue1.offer(new int[]{i, 0}); }
        for (int j = 0; j < m; j++) { firstBorder[0][j] = true; queue1.offer(new int[]{0, j}); }

        // Seed right column + bottom row into queue2
        for (int i = 0; i < n; i++) { secondBorder[i][m-1] = true; queue2.offer(new int[]{i, m-1}); }
        for (int j = 0; j < m; j++) { secondBorder[n-1][j] = true; queue2.offer(new int[]{n-1, j}); }

        bfs(firstBorder, queue1);
        bfs(secondBorder, queue2);

        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                if (firstBorder[i][j] && secondBorder[i][j])
                    System.out.println(i + " " + j);
    }
}
```
